service:
	mysql-server:
		image: MySQL:8.0
		container_name: MySQL-API
		environment: 
			MYSQL_DATABASE: API_Ujikom
			MYSQL_USER: api_ujikom
			MYSQL_PASSWORD: api_ujikom
			MYSQL_ROOT_PASSWORD: passwordroot
		ports: 
			- "3606:3606"
		networks:
			- API_UJIKOM
		volumes:
			- db_data:/var/lib/MySQL

	app-Laravel:
		build:
			context: ./backend
			dockerfile: Dockerfile
		container_name: laravel-api
		working_dir: /var/www
		ports:
			- "8000:8000"
		volumes: 
			- ./backend:/var/www
		command: sh -c "php artisan storage:link; php artisan serve --host=0.0.0.0 --ports=8000"
		depends_on:
			- mysql-server
		networks:
			- API-UJIKOM

	scheduler:
		build:
			context: ./backend
			dockerfile: Dockerfile
		container_name: laravel-scheduler
		working_dir: /var/www
		volumes:
			- ./backend:/vaar/www
		command: php artisan schedule:work
		depends_on:
			- MySQL-server
		networks:
			- API-UJIKOM

	phpMyadmin:
		image: phpMyAdmin
		container_name: PhpMyadmin-API
		environment: 
			PMA_HOST: myql-server
		ports: 
			- "8081:80"
		depends_on: 
			- mysql-server
		networks:
			- API-UJIKOM

		db_data:
	networks:
